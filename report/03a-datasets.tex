\subsection{Preprocessing}\label{sec:prep}
% Sources: experiments/07_usgs_earthquakes_2023_eda.ipynb, section 12;
% experiments/10_data_gov_hdb_resale_eda.ipynb, section 9; src/sc4020/datasets/preprocess_dataset.py (jitter_ties).
\begin{itemize}\setlength{\itemsep}{0pt}
\item \textbf{Latitude, longitude (D2)} $\to$ haversine distance for density methods, 3D unit-sphere
points for K-means.
\item \textbf{Latitude, longitude (D3)} $\to$ kilometres, projected about the mean latitude.
\item \textbf{Features in raw units (D1, D4, D6, D8)} $\to$ mean 0, standard deviation 1.
\item \textbf{Stepped values (D8, density methods only)} $\to$ uniform jitter of up to half a step:
$\pm1.5$ storeys, $\pm0.5$\,m$^2$ floor area, $\pm\frac{1}{24}$ year of lease.
\end{itemize}
