\subsection{Preprocessing}\label{sec:prep}
% Source: src/sc4020/datasets/load_dataset.py, preprocess_dataset.py (to_unit_vectors, project_km, jitter_ties).
\begin{itemize}\setlength{\itemsep}{0pt}
\item \textbf{Latitude, longitude (D2)} $\to$ 3D points on the unit sphere, so straight-line distance
follows the Earth's surface and the $180^\circ$ meridian is no border.
\item \textbf{Latitude, longitude (D3)} $\to$ kilometres, projected about the mean latitude.
\item \textbf{Features in raw units (D1, D4, D6, D8)} $\to$ mean 0, standard deviation 1.
\item \textbf{Images (D5)} $\to$ pixels scaled to $[0,1]$, then 50 principal components.
\item \textbf{Stepped values (D8)} $\to$ uniform jitter of up to half a step, so tied values do not form
artificial dense spots: $\pm1.5$ storeys, $\pm0.5$\,m$^2$ floor area, $\pm\frac{1}{24}$ year of lease.
\end{itemize}
